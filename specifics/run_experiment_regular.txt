SERVER RESPONSE (Protocol 2.1 - Binary Acquisition Data)

// Step 1: Server sends metadata JSON
{
  "cmd": "run_experiment",
  "session_id": "ses_001",
  "ok": true,
  "config_log": [ "gen 0 drive frequency: 150.0 MHz", "..." ],
  "adc_metadata": {
    "0": {  // ADC index (string key)
      "dtype": "complex64",           // or "complex128" for accumulated mode
      "shape": [1000, 2048],          // [shots, samples] or [samples] if accumulated
      "byte_order": "little",
      "size_bytes": 16384000          // Total bytes for this ADC's data
    }
  },
  "debug_timing": {
    "total_server_time_ms": 150.5,
    "server_encode_ms": 12.3,
    "payload_bytes": 16384000
  },
  "error": null
}

// Step 2: Server sends binary frames (immediately after metadata JSON)
// For each ADC in metadata:
//
// Frame format:
// [4-byte ADC index (uint32 big-endian)]
// [4-byte data length in bytes (uint32 big-endian)]
// [raw numpy complex array bytes from .tobytes()]
//
// Example for ADC 0:
// Header: 8 bytes = [adc_idx=0][length=16384000]
// Data: 16384000 bytes = raw complex64 array (1000 shots × 2048 samples × 8 bytes/sample)
//
// Client reconstruction:
// arr = np.frombuffer(data_bytes, dtype='complex64')
// arr = arr.reshape([1000, 2048])  // shape from metadata
//
// Result: Complex numpy array where:
//   - Real part = I component
//   - Imaginary part = Q component
//   - Access: np.real(arr), np.imag(arr), or np.abs(arr) for amplitude
