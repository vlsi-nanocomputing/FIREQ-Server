// Sweep experiments return a single header with all point metadata,
// then stream binary frames only (in batches of points).

// HEADER MESSAGE (sent once at the beginning):

{
  "type": "sweep_header",
  "sweep_id": "example_sweep_001",
  "batch_size": 10,
  "n_points": 125,
  "points": [
    {
      "point_index": 0,
      "variables": {"DRV_FREQ": 100.0, "RO_GAIN": 0.1, ...}  // Resolved variable values
    },
    {
      "point_index": 1,
      "variables": {"DRV_FREQ": 105.0, "RO_GAIN": 0.2, ...}
    }
    // ... up to all points in the sweep
  ],
  "adc_metadata": {
    "0": {
      "dtype": "complex64",
      "shape": [32, 1024]                // [shots, samples] or [samples] if accumulated
    }
  }
}

// BINARY STREAM (repeated in batches):
// For each point (in order, batched):
//   For each ADC in the sweep:
//     [4-byte ADC index (uint32 big-endian)]
//     [4-byte data length in bytes (uint32 big-endian)]
//     [raw numpy complex array bytes]
//
// Example for batch with 2 points, each with 1 ADC:
// Point 0, ADC 0: [adc_idx=0][length=262144][262144 bytes of complex64 data]
// Point 1, ADC 0: [adc_idx=0][length=262144][262144 bytes of complex64 data]


// FINAL STATUS MESSAGE (after all batches):
{
  "type": "sweep_status",
  "sweep_id": "example_sweep_001",
  "ok": true,
  "total_points": 125,                  // Total points in sweep (cartesian product of all variables)
  "completed_points": 125,
  "debug_timing": {
    "total_server_time_ms": 5234.2,
    "server_encode_ms": 234.5,
    "payload_bytes": 32768000
  },

}
